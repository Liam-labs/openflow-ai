# API handlers that answer without a DevFlow token or a local caller, on purpose. One per line: METHOD /route  # why.
# scripts/security-routes.mjs --fail-on-open fails for any other handler without a guard.
GET /api/setup/status            # first-run check used before any token exists; without a token or trusted local request it says only which steps are done (no entered data)
POST /api/slack/events           # Slack signs every request; the handler checks the HMAC with the signing secret
POST /api/teams-bot/messages     # Bot Framework token checked by verifyBotToken
POST /api/teams-bot/lifecycle    # Graph subscription handshake; lifecycle events are only logged
GET /api/teams-bot/lifecycle     # fixed health reply
POST /api/teams-bot/notifications  # each notification must carry the secret we gave Graph (clientState); handshake echoes its own token
POST /api/voice/enrollment       # both branches refuse before any work; which guard applies depends on the "operation" in the JSON body, so the body is read first
POST /api/voice/turn/cancel      # same: the guard is chosen from the body's "action" (companion token or normal auth), then refuses before any work
